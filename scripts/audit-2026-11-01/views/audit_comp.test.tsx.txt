import { render, screen, waitFor } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { useState } from 'react'
import { afterEach, beforeEach, describe, expect, test, vi } from 'vitest'
import type { Row, ViewState } from '@/api/types'
import { objectSummary, propertyDef, row, typeDef } from '@/test/mockApi'

vi.mock('@/api/client', async (orig) => ({
  ...(await orig<typeof import('@/api/client')>()),
  api: (await import('@/test/mockApi')).mockApi(),
}))

import { api } from '@/api/client'
import { useSettings } from '@/settings/useSettings'
import { DataView } from './DataView'
import { FilterEditor } from './FilterEditor'
import { emptyFilterGroup } from './model'
import { propertyChoices } from './properties'

const mocked = vi.mocked(api)

const PAGE = typeDef({
  id: 'page',
  propertyDefinitions: [
    propertyDef({ id: 'title', type: 'entity_title', name: 'Title' }),
    propertyDef({ id: 'amount', type: 'number', name: 'Amount' }),
    propertyDef({ id: 'due', type: 'datetime', name: 'Due' }),
  ],
})
const mk = (path: string, values: Row['values'] = {}, version = 'v1'): Row => row({ summary: objectSummary({ path, type: 'page' }), values, version })

function Harness({ rows, patchRow }: { rows: Row[]; patchRow?: (p: string, r: Row) => void }) {
  const [state, setState] = useState<ViewState>({ viewDefinition: { dataViewLayout: 'table' }, filterGroup: emptyFilterGroup(), sorts: [], groupBy: null })
  return <DataView rows={rows} types={[PAGE]} scope={{ typeIds: ['page'] }} state={state} onState={setState} layouts={['table']} patchRow={patchRow} />
}

beforeEach(() => vi.clearAllMocks())
afterEach(() => vi.useRealTimers())

describe('audit: table number cell', () => {
  test('D1 after a failed write, retyping the same number is never saved again', async () => {
    const user = userEvent.setup()
    mocked.updateProperties.mockRejectedValueOnce(new Error('disk full'))
    render(<Harness rows={[mk('Pages/A.md', { amount: 1 }, 'v7')]} />)
    const input = screen.getByRole('textbox', { name: 'Amount' })
    await user.clear(input)
    await user.type(input, '5')
    await user.tab()
    await waitFor(() => expect(mocked.updateProperties).toHaveBeenCalledTimes(1))
    await screen.findByText('disk full')
    // The cell shows the stored value again (1); the user types 5 once more.
    expect(input).toHaveValue('1')
    await user.clear(input)
    await user.type(input, '5')
    await user.tab()
    await new Promise((r) => setTimeout(r, 400))
    // Second attempt: nothing is sent, the edit is silently dropped.
    expect(mocked.updateProperties).toHaveBeenCalledTimes(1)
  })
})

describe('audit: filter editor date default', () => {
  test('E1 "Specific date" defaults to the UTC day, not the Time zone setting day', async () => {
    useSettings.setState({ settings: { ...useSettings.getState().settings, timeZone: 'America/Los_Angeles' } })
    vi.useFakeTimers({ shouldAdvanceTime: true, now: new Date('2026-11-01T03:00:00Z') }) // Oct 31, 8 PM in LA
    const user = userEvent.setup({ advanceTimers: vi.advanceTimersByTime })
    const onChange = vi.fn()
    const choices = propertyChoices([PAGE])
    const group = { ...emptyFilterGroup(), elements: [{ type: 'filter' as const, propertyId: 'due', option: 'is' }] }
    try {
      render(<FilterEditor group={group} onChange={onChange} choices={choices} types={[PAGE]} collections={[]} />)
      await user.click(screen.getByRole('button', { name: 'Date' }))
      await user.click(await screen.findByRole('menuitemradio', { name: 'Specific date' }))
      const f = onChange.mock.calls[0][0].elements[0]
      expect(f.referenceValue).toEqual({ exactDate: '2026-10-31' })
    } finally {
      useSettings.setState({ settings: { ...useSettings.getState().settings, timeZone: null } })
    }
  })
})
